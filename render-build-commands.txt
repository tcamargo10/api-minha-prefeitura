# Comandos de build para o Render.com
# Estes comandos devem ser executados na ordem especificada

# 1. Instalar dependências
npm ci

# 2. Gerar cliente Prisma
npx prisma generate

# 3. Compilar TypeScript
npx tsc

# 4. Verificar se o build foi bem-sucedido
ls -la dist/

# 5. Comando de início
node dist/index.js
